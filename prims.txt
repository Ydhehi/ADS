
#include <stdio.h>
#define INF 9999

int main()
{
    int n;

    printf("enter no of nodes:");
    scanf("%d", &n);

    int c[10][10], vi[10] = {0};

    printf("enter adj matrix:");

    for(int i=0; i<n; i++)
        for(int j=0; j<n; j++)
            scanf("%d", &c[i][j]);

    vi[0] = 1;

    int edges = 0;

    printf("edges in mst:\n");

    while(edges < n-1)
    {
        int min = INF, a = -1, b = -1;

        for(int i=0; i<n; i++)
        {
            if(vi[i])
            {
                for(int j=0; j<n; j++)
                {
                    if(!vi[j] && c[i][j]!=0 && c[i][j]<min)
                    {
                        min = c[i][j];
                        a = i;
                        b = j;
                    }
                }
            }
        }

        if(b == -1)
        {
            printf("MST not possible");
            return 0;
        }

        printf("%d--%d(weight=%d)\n", a, b, min);

        vi[b] = 1;
        edges++;
    }

    return 0;
}
