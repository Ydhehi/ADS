#include <stdio.h>

int main()
{
    int n, g[10][10], v[10] = {0};
    int Q[10], f = 0, r = -1;

    printf("enter no of nodes:");
    scanf("%d", &n);

    printf("enter adjacency matrix:\n");

    for(int i = 0; i < n; i++)
        for(int j = 0; j < n; j++)
            scanf("%d", &g[i][j]);

    int start;

    printf("enter starting node:");
    scanf("%d", &start);

    v[start] = 1;
    Q[++r] = start;

    printf("BFS order:");

    while(f <= r)
    {
        int node = Q[f++];
        printf("%d ", node);

        for(int i = 0; i < n; i++)
        {
            if(g[node][i] == 1 && v[i] == 0)
            {
                v[i] = 1;
                Q[++r] = i;
            }
        }
    }

    return 0;
}
