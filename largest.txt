#include <stdio.h>
int largest(int a[],int n)
{
    int i;
    int max=0;
    for(i=0;i<n;i++)
        {
            if(a[i]>max)
            {
                max=a[i];
            }
        }
    return max;
}
void main()
{
    int a[10],i,n,result;
    printf("enter number of elements:");
    scanf("%d",&n);
    printf("enter the elements:");
    for(i=0;i<n;i++)
        {
           scanf("%d",&a[i]);  
        }
    printf("array elements are:");
    for(i=0;i<n;i++)
    {
        printf("%d",a[i]);
    }
    result=largest(a,n);
    printf("largest element=%d",result);
}
