#include <stdio.h>
#include <stdlib.h>
struct node
{
    int data;
    struct node *left,*right;
};
struct node *createNode(int value)
{
    struct node *newNode;
    newNode=(struct node*)malloc(sizeof(struct node));
    newNode->data=value;
    newNode->left=NULL;
    newNode->right=NULL;
    return newNode;
}
struct node *insert(struct node *root,int value)
{
    if(root==NULL)
    {
        return createNode(value);
    }
    if(value<root->data)
    {
        root->left=insert(root->left,value);
    }
    else if(value>root->data)
    {
        root->right=insert(root->right,value);
    }
       return root;                 
}
       void preorder(struct node *root)
       {
            if(root!=NULL)
            {
                printf("%d",root->data);
                preorder(root->left);
                preorder(root->right);
            }
        
       }
int main()
{
    struct node *root=NULL;
    root=insert(root,7);
    insert(root,5);
    insert(root,1);
    insert(root,8);
    insert(root,3);
    insert(root,6);
    printf("Preorder traversal:");
    preorder(root);
    return 0;
}
